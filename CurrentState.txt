Task 1.3: End-to-End Testing - 50 Queries (1 hour)
Create test script: test-performance.ps1
powershell# Performance Test Script
$baseUrl = "http://localhost:8080"
$token = "YOUR_JWT_TOKEN"  # Get from login

$queries = @(
    "What equipment do employees get?",
    "How many vacation days?",
    "What is the vacation carryover policy?",
    "How to request time off?",
    "What is git rebase?",
    "How to push to remote?",
    "What is the company equipment policy?",
    "Who is eligible for vacation?",
    "How to create a git branch?",
    "What happens to equipment on termination?"
)

$results = @()

Write-Host "🚀 Starting performance test (50 queries)..." -ForegroundColor Green
Write-Host ""

for ($i = 1; $i -le 50; $i++) {
    $query = $queries[$i % $queries.Length]

    $body = @{
        query = $query
        topK = 5
    } | ConvertTo-Json

    $start = Get-Date

    try {
        $response = Invoke-RestMethod -Uri "$baseUrl/api/search" `
            -Method POST `
            -Headers @{
                "Authorization" = "Bearer $token"
                "Content-Type" = "application/json"
            } `
            -Body $body

        $elapsed = ((Get-Date) - $start).TotalMilliseconds

        $result = [PSCustomObject]@{
            Query = $i
            Time = [math]::Round($elapsed, 0)
            Success = $true
            Answer = $response.answer.Substring(0, [Math]::Min(50, $response.answer.Length))
        }

        Write-Host "Query $i`: $($elapsed)ms - " -NoNewline
        Write-Host "✅" -ForegroundColor Green

    } catch {
        $elapsed = ((Get-Date) - $start).TotalMilliseconds
        $result = [PSCustomObject]@{
            Query = $i
            Time = [math]::Round($elapsed, 0)
            Success = $false
            Answer = $_.Exception.Message
        }
        Write-Host "Query $i`: $($elapsed)ms - " -NoNewline
        Write-Host "❌" -ForegroundColor Red
    }

    $results += $result
    Start-Sleep -Milliseconds 100  # Small delay between requests
}

Write-Host ""
Write-Host "📊 PERFORMANCE SUMMARY" -ForegroundColor Cyan
Write-Host "======================" -ForegroundColor Cyan

$successful = $results | Where-Object { $_.Success -eq $true }
$failed = $results | Where-Object { $_.Success -eq $false }

Write-Host "Total Queries: 50"
Write-Host "Successful: $($successful.Count)" -ForegroundColor Green
Write-Host "Failed: $($failed.Count)" -ForegroundColor Red
Write-Host ""

if ($successful.Count -gt 0) {
    $times = $successful | Select-Object -ExpandProperty Time
    Write-Host "Latency Statistics:" -ForegroundColor Yellow
    Write-Host "  Min: $($times | Measure-Object -Minimum | Select-Object -ExpandProperty Minimum)ms"
    Write-Host "  Max: $($times | Measure-Object -Maximum | Select-Object -ExpandProperty Maximum)ms"
    Write-Host "  Avg: $([math]::Round(($times | Measure-Object -Average | Select-Object -ExpandProperty Average), 0))ms"
    Write-Host "  Median: $($times | Sort-Object | Select-Object -Index ([math]::Floor($times.Count / 2)))ms"
}

Write-Host ""
Write-Host "✅ Test complete! Results saved to test-results.csv"
$results | Export-Csv -Path "test-results.csv" -NoTypeInformation

## Prerequisites

- Java 21
- PostgreSQL 16+
- Qdrant (standalone or Docker)

## Quick Start

### 1. Database Setup

```bash
export GROQ_API_KEY=your-groq-api-key  # Optional, for remote LLM
```

### 4. Build & Run
```bash

mvn clean package -DskipTests

# Run
java -jar target/sapphire-ai-service-1.0.0.jar
```

### 5. Verify

## Configuration Profiles

### Local Development
```bash
mvn spring-boot:run -Dspring.profiles.active=local
```

### Production
```bash
java -jar sapphire-ai-service.jar --spring.profiles.active=prod
```

## Performance Tuning

### Expected Metrics (After Warmup)
- First query: ~5 seconds (model loading)
- Subsequent queries: 300-500ms
- Concurrent users: Up to 10 simultaneous searches

### Troubleshooting

**Slow first query:**
- Normal - embedding model loads on first request
- Warmup runs automatically on startup
- Check logs for "System warmup complete"

**Qdrant connection failed:**
- Verify Qdrant is running: `curl http://localhost:6333`
- Check port: Should be 6334 for gRPC
- Restart Qdrant if needed

**JWT errors:**
- Ensure `sapphire.security.jwt.secret` is set
- Must be 64+ characters, alphanumeric only
- No dashes or special characters

## Production Checklist

- [ ] Change JWT secret to random 64+ char string
- [ ] Set strong database password
- [ ] Configure GROQ_API_KEY (if using remote LLM)
- [ ] Enable HTTPS (not included in MVP)
- [ ] Configure firewall (allow 8080, block 6333/6334)
- [ ] Set up database backups
- [ ] Monitor logs in `/var/log/sapphire-ai/`

Task 2.1: List Documents Endpoint (1 hour)
File: src/main/java/com/sapphireai/controller/DocumentController.java
Add this method:
java@GetMapping
public ResponseEntity<Page<DocumentResponse>> listDocuments(
        Authentication authentication,
        @RequestParam(defaultValue = "0") int page,
        @RequestParam(defaultValue = "20") int size,
        @RequestParam(required = false) ProcessingStatus status
) {
    var user = (UserDetailsImpl) authentication.getPrincipal();

    Pageable pageable = PageRequest.of(page, size, Sort.by("createdAt").descending());

    Page<ProcessedDocument> documents;
    if (status != null) {
        documents = documentRepository.findByOrganizationIdAndStatus(
            user.getOrganizationId(), status, pageable
        );
    } else {
        documents = documentRepository.findByOrganizationId(
            user.getOrganizationId(), pageable
        );
    }

    Page<DocumentResponse> response = documents.map(doc -> DocumentResponse.builder()
            .id(doc.getId())
            .fileName(doc.getFileName())
            .contentType(doc.getContentType())
            .fileSize(doc.getFileSize())
            .status(doc.getStatus())
            .documentType(doc.getDocumentType())
            .totalChunks(doc.getTotalChunks())
            .processingTimeMs(doc.getProcessingTimeMs())
            .createdAt(doc.getCreatedAt())
            .build());

    return ResponseEntity.ok(response);
}

Task 2.2: Delete Document Endpoint (1 hour)
Add to DocumentController.java:
java@DeleteMapping("/{documentId}")
public ResponseEntity<Void> deleteDocument(
        @PathVariable String documentId,
        Authentication authentication
) {
    var user = (UserDetailsImpl) authentication.getPrincipal();

    ProcessedDocument document = documentRepository.findById(documentId)
            .orElseThrow(() -> new RuntimeException("Document not found"));

    // Verify ownership
    if (!document.getOrganizationId().equals(user.getOrganizationId())) {
        return ResponseEntity.status(HttpStatus.FORBIDDEN).build();
    }

    // Delete chunks
    List<DocumentChunk> chunks = chunkRepository.findByDocumentId(documentId);
    chunkRepository.deleteAll(chunks);

    // Delete vectors
    vectorStoreService.deleteDocumentVectors(documentId);

    // Delete file
    try {
        if (document.getFilePath() != null) {
            Files.deleteIfExists(Paths.get(document.getFilePath()));
        }
    } catch (IOException e) {
        log.warn("Failed to delete file: {}", document.getFilePath(), e);
    }

    // Delete document record
    documentRepository.delete(document);

    log.info("Document deleted: {}", documentId);
    return ResponseEntity.noContent().build();
}
